cmake_minimum_required(VERSION 3.16)
project(LibYuvSharpNative C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The packaged native library requires Windows x64.")
endif()
find_package(JPEG REQUIRED)
set(LIBYUV_DISABLE_JPEG OFF CACHE BOOL "Build without JPEG" FORCE)
set(UNIT_TEST OFF CACHE BOOL "Build upstream tests" FORCE)
add_compile_definitions(LIBYUV_BUILDING_SHARED_LIBRARY)
add_subdirectory("${LIBYUV_SOURCE_DIR}" libyuv)
set_target_properties(yuv_shared PROPERTIES OUTPUT_NAME libyuv_internal PREFIX "")
if(MINGW)
  target_link_options(yuv_shared PRIVATE -static-libgcc -static-libstdc++)
endif()
