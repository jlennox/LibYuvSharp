cmake_minimum_required(VERSION 3.16)
project(LibYuvSharpNative C CXX)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The packaged native libraries require x64 or ARM64.")
endif()
include(CheckCSourceCompiles)
if(LIBYUV_RID MATCHES "-arm64$")
  check_c_source_compiles("#if !defined(__aarch64__) && !defined(_M_ARM64)\n#error ARM64 compiler required\n#endif\nint main(void) { return 0; }" LIBYUV_COMPILER_ARM64)
  if(NOT LIBYUV_COMPILER_ARM64)
    message(FATAL_ERROR "The selected compiler does not target ARM64")
  endif()
elseif(LIBYUV_RID MATCHES "-x64$")
  check_c_source_compiles("#if !defined(__x86_64__) && !defined(_M_X64)\n#error x64 compiler required\n#endif\nint main(void) { return 0; }" LIBYUV_COMPILER_X64)
  if(NOT LIBYUV_COMPILER_X64)
    message(FATAL_ERROR "The selected compiler does not target x64")
  endif()
endif()
find_package(JPEG REQUIRED)
set(LIBYUV_DISABLE_JPEG OFF CACHE BOOL "Build without JPEG" FORCE)
set(UNIT_TEST OFF CACHE BOOL "Build upstream tests" FORCE)
add_subdirectory("${LIBYUV_SOURCE_DIR}" libyuv)
target_compile_definitions(yuv_common_objects PRIVATE LIBYUV_BUILDING_SHARED_LIBRARY)
set_target_properties(yuv_shared PROPERTIES OUTPUT_NAME libyuv_internal PREFIX "")
if(MINGW)
  target_link_options(yuv_shared PRIVATE -static-libgcc -static-libstdc++)
endif()
if(ANDROID)
  target_link_options(yuv_shared PRIVATE -Wl,-z,max-page-size=16384)
endif()

add_executable(libyuv_smoke smoke.cc)
target_include_directories(libyuv_smoke PRIVATE "${LIBYUV_SOURCE_DIR}/include")
target_compile_definitions(libyuv_smoke PRIVATE LIBYUV_USING_SHARED_LIBRARY)
target_link_libraries(libyuv_smoke PRIVATE yuv_shared)
if(MINGW)
  target_link_options(libyuv_smoke PRIVATE -static-libgcc -static-libstdc++)
endif()
set_target_properties(libyuv_smoke PROPERTIES BUILD_RPATH "$ORIGIN/libyuv;$ORIGIN")
add_custom_command(TARGET libyuv_smoke POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:yuv_shared>" "$<TARGET_FILE_DIR:libyuv_smoke>")
enable_testing()
add_test(NAME native_smoke COMMAND libyuv_smoke "${CMAKE_CURRENT_SOURCE_DIR}/../LibYuvSharp.Test/solid-color.jpg")
